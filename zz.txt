async function validateResult(req,res,next)
{
    const result = validateRules(req);
    if(!result.isEmpty()){
        return res.status(400).json({error : result.array()})
    }
    next()
}

const validateRules = [
    body("userName")
        .isString()
        .withMessage("User name must be a string)
        .isLength({min:5 , max:30})
        .withMessage("User name length must be between 5 and 30")

    body("email")
        .isEmail()
        .withMessage("Correct email required...)
    validateResult
]
